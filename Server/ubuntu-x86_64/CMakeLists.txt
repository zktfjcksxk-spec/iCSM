cmake_minimum_required(VERSION 3.22)
option(BUILD_M6 "Build the native Metal scene and Panorama backend" OFF)
if(BUILD_M6)
  set(BUILD_M5 ON)
endif()
project(icsm_ubuntu_x86_64 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Linux required")
endif()
set(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/source" CACHE PATH "2019 source")
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,defs")
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN")
add_library(source_platform INTERFACE)
if(BUILD_M6)
  target_compile_definitions(source_platform INTERFACE SOURCE_NATIVE_METAL)
  target_include_directories(source_platform INTERFACE "${CMAKE_CURRENT_LIST_DIR}/compat/metal")
endif()
target_compile_definitions(source_platform INTERFACE GNUC POSIX _POSIX LINUX _LINUX COMPILER_GCC DEDICATED NO_STEAM NO_STEAM_GAMECOORDINATOR)
target_compile_definitions(source_platform INTERFACE "$<$<CONFIG:Debug>:DEBUG;_DEBUG>")
target_include_directories(source_platform INTERFACE "${SOURCE_ROOT}/public" "${SOURCE_ROOT}/common"
  "${SOURCE_ROOT}/public/tier0" "${SOURCE_ROOT}/public/tier1" "${SOURCE_ROOT}/thirdparty/sse2neon")
target_compile_definitions(source_platform INTERFACE RAD_TELEMETRY_DISABLED CSTRIKE15
  _DLL_EXT=.so DX_TO_GL_ABSTRACTION GL_GLEXT_PROTOTYPES
  SSE2NEON_PRECISE_MINMAX=1 SSE2NEON_PRECISE_DIV=1 SSE2NEON_PRECISE_SQRT=0)
# This pinned header uses native arm64 FSQRT with PRECISE_SQRT=0. Its optional
# reciprocal approximation loses infinity/signed-zero behavior on arm64.
# Original source_posix_base.vpc uses hidden symbol visibility. VGUI maps
# contain module-local inline statics; exporting them coalesces registrations
# across Mach-O images and breaks console input and widget styling.
if(BUILD_M5)
  target_compile_options(source_platform INTERFACE -fvisibility=hidden -mcx16)
endif()
target_compile_options(source_platform INTERFACE -fno-strict-aliasing -Wno-register -Wno-invalid-source-encoding -Wno-c++11-narrowing)

# POSIX file list from the local tier0/tier0.inc, checked against Kisak CMake.
set(TIER0_FILES assert_dialog commandline cpu cpumonitoring cputopology cpu_posix
  dbg fasttimer logging mem mem_helpers memdbg memvirt memstd memvalidate
  dynfunction minidump tier0_strtools miniprofiler perfstats platform_posix
  platform_independent platwindow pme_posix progressbar security stackstats
  stacktools systeminformation threadtools tslist vatoms vprof vprof_telemetry3
  vtuneinterface win32consoleio)
list(TRANSFORM TIER0_FILES PREPEND "${SOURCE_ROOT}/tier0/")
list(TRANSFORM TIER0_FILES APPEND ".cpp")
add_library(tier0 SHARED ${TIER0_FILES} "${SOURCE_ROOT}/unitlib/unitlib.cpp")
target_link_libraries(tier0 PRIVATE source_platform)
target_compile_definitions(tier0 PRIVATE TIER0_DLL_EXPORT RAD_TELEMETRY_DISABLED CROSS_PLATFORM_VERSION=1)
target_include_directories(tier0 PRIVATE "${SOURCE_ROOT}/tier0" "${SOURCE_ROOT}/public/tier0" "${SOURCE_ROOT}/public/tier1")
target_include_directories(tier0 PRIVATE "${CMAKE_CURRENT_LIST_DIR}/vendor/SDL2/include")
add_executable(tier0_probe probes/tier0_probe.cpp)
target_link_libraries(tier0_probe PRIVATE source_platform tier0)
set_target_properties(tier0_probe PROPERTIES BUILD_RPATH "$ORIGIN")
enable_testing()
add_test(NAME tier0_native_probe COMMAND tier0_probe "$<TARGET_FILE:tier0>")
set_tests_properties(tier0_native_probe PROPERTIES TIMEOUT 30)
add_executable(tier0_loader probes/tier0_loader.cpp)
target_link_libraries(tier0_loader PRIVATE source_platform)
add_test(NAME tier0_independent_loader COMMAND tier0_loader "$<TARGET_FILE:tier0>")
set_tests_properties(tier0_independent_loader PROPERTIES TIMEOUT 30)
target_link_libraries(source_platform INTERFACE dl pthread rt)
target_compile_options(source_platform INTERFACE -fvisibility=hidden -mcx16)
include(cmake/foundation.cmake)
foreach(probe math thread resource)
  add_executable(${probe}_probe probes/${probe}_probe.cpp)
  target_link_libraries(${probe}_probe PRIVATE source_platform tier1 mathlib vstdlib interfaces tier0)
  set_target_properties(${probe}_probe PROPERTIES BUILD_RPATH "$ORIGIN")
endforeach()
add_dependencies(resource_probe filesystem_stdio)
add_test(NAME math_native_probe COMMAND math_probe)
add_test(NAME thread_native_probe COMMAND thread_probe)
set(GAME_RESOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../game-assets/csgo" CACHE PATH "Shared original game resources")
add_test(NAME resource_native_probe COMMAND resource_probe "$<TARGET_FILE:filesystem_stdio>"
  "$<TARGET_FILE:vstdlib>" "${GAME_RESOURCE_ROOT}" "${CMAKE_CURRENT_LIST_DIR}/provenance/m1/resource-samples.tsv")
set_tests_properties(math_native_probe thread_native_probe resource_native_probe PROPERTIES TIMEOUT 60)
install(TARGETS tier0 vstdlib filesystem_stdio tier0_loader tier0_probe math_probe thread_probe resource_probe
  RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
install(TARGETS interfaces tier1 mathlib tier2 vpklib ARCHIVE DESTINATION lib)

include(cmake/physics.cmake)

option(BUILD_M3 "Build the native headless server chain" OFF)
option(BUILD_PRIVATE_INVENTORY "Synchronize per-player private inventory" ON)
option(BUILD_M4 "Build the native OpenGL client chain" OFF)
option(BUILD_M5 "Build the original Panorama offline client chain" OFF)
if(BUILD_M5)
  set(BUILD_M4 ON)
  if(CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "27.0")
    message(FATAL_ERROR "The frozen M5 native libraries require macOS 27.0; configure with -DCMAKE_OSX_DEPLOYMENT_TARGET=27.0")
  endif()
  include(cmake/m5-native.cmake)
  target_compile_definitions(source_platform INTERFACE PANORAMA_ENABLE)
endif()
if(BUILD_M3 OR BUILD_M4)
  include(cmake/headless.cmake)
endif()
if(BUILD_PRIVATE_INVENTORY)
  include("${CMAKE_CURRENT_LIST_DIR}/private-inventory.cmake")
endif()

  enable_language(OBJCXX)
  include(cmake/client.cmake)
endif()
if(BUILD_M5)
  include(cmake/panorama.cmake)
endif()

  set_tests_properties(metal_original_ui_shader_probe PROPERTIES TIMEOUT 60)
